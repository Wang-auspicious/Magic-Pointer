SvgCircle(cx: number, cy: number, r: number, fill?: string, stroke?: string, strokeWidth?: number, opacity?: number) — Circle. Numeric properties accept expressions using reactive $variables.
SvgRect(x: number, y: number, width: number, height: number, fill?: string, stroke?: string, strokeWidth?: number, opacity?: number, rx?: number) — Rectangle; rx optionally rounds the corners.
SvgLine(x1: number, y1: number, x2: number, y2: number, stroke?: string, strokeWidth?: number, opacity?: number) — Line between two points; distinct from chart components.
SvgPath(d: string, fill?: string, stroke?: string, strokeWidth?: number, opacity?: number) — SVG path commands such as M, L, C, Q, A and Z. No markup or URLs.
SvgText(text: string, x: number, y: number, size?: number, fill?: string, anchor?: string) — Plain text in the drawing. anchor is start, middle or end. Use expressions for live numeric labels.
SvgGroup(children: (SvgCircle | SvgRect | SvgLine | SvgPath | SvgText | SvgGroup)[], transform?: string, opacity?: number) — Group shapes with translate, rotate, scale, skewX, skewY or matrix. Groups may nest.
Svg(children: (SvgCircle | SvgRect | SvgLine | SvgPath | SvgText | SvgGroup)[], viewBox: string, width?: number, height?: number, label?: string) — Responsive local drawing. viewBox is four numbers: minX minY width height. Dimensions are optional pixels; label describes the diagram. Paint accepts CSS colors, currentColor or none; URLs are not supported.
